Modelo de dados

imutável no repo · mutável no Turso (2 tabelas, com slots pra engrossar) · ✓ implementado em db/schema.sql

-- Turso · estado mutável
CREATE TABLE freshness (
  ficha_slug     TEXT PRIMARY KEY,
  estado         TEXT,     -- fresco | frio
  calculado_em   INTEGER,  -- epoch
  fonte          TEXT,
  previsao_bruta TEXT       -- transparência
);
CREATE TABLE confirmacoes (
  id          INTEGER PRIMARY KEY,
  ficha_slug  TEXT,
  criado_em   INTEGER,
  tipo        TEXT -- slot: 2 sinais depois
);
